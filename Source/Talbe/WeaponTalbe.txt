# ПОЛЯ (по порядку):
#   key            — внутренний идентификатор вида оружия (sword/dagger/...)
#   name           — отображаемое имя (для UI/HUD), кириллица допустима
#   category       — папка спрайтов на сервере (Source/Weapon/<category>/...)
#   throwSpeed     — начальная скорость брошенного оружия (px/кадр). Определяет
#                    дальность броска — чем больше, тем дальше летит.
#   scale          — базовый масштаб клинка относительно эталонного меча (1.0 = как меч).
#                    Это же поле = "длина" оружия.
#   dmgBase        — базовый урон брошенного оружия при попадании (константа)
#   dmgPerSpeed    — множитель урона за скорость полёта (урон = dmgBase + скорость*dmgPerSpeed)
#   maxDmgPercent  — потолок урона за ОДНО попадание, в долях от ТЕКУЩЕГО HP цели
#   spinMin/spinMax— угловая скорость вращения брошенного/выбитого оружия в полёте.
#                    0/0 = летит прямо без вращения ("полёт прямой").
#   weight         — вес. Влияет на скорость замаха/поворота клинка в руке И на
#                    скорость передвижения персонажа (тяжёлое оружие = медленнее и то, и то).
#   staminaMult    — "Отдых": множитель расхода стамины за замах/флик этим оружием
#                    относительно базового (sl-stamswing / sl-stamflick).
#   cutMult        — "Режущий": множитель урона для АТАК С КРУЧЕНИЕМ (обычный замах).
#   pierceMult     — "Колющий": множитель урона для АТАК ПО ПРЯМОЙ (укол, окно ~30мс).
#   collision      — "full" = коллайдер урона на всю длину клинка,
#                    "tip"  = урон наносит только наконечник (~последние 30% длины);
#                    коллайдер-БЛОК (клинок о клинок) всегда на всю длину независимо от этого поля.
#   flags          — спецспособности через запятую:
#                    disarm    — при нанесении урона всегда выбивает оружие у цели
#                    knockback — сильный доп. нокбэк по телу/щиту цели при попадании
#   chargeTime     — [ТОЛЬКО ЖЕЗЛ] время накопления заряда перед выстрелом, сек.
#                    Для остального оружия не используется, можно оставлять пустым.
#   lmbStaminaMult — множитель стоимости стамины ЛКМ-удара (сверху общего % из
#                    настроек "Стам. ЛКМ (начало)" / sl-lmbcost, по умолчанию 50%).
#                    1.0 = обычная стоимость (50%), 0.5 = вдвое дешевле (25%,
#                    как у рапиры). Если не указано — 1.0.
